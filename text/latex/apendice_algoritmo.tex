\refstepcounter{algorithm}%
\noindent\textbf{Algoritmo~\thealgorithm}: arranjo sequencial do corte do OpenRecLink e GZ-CMD v3.1, resultado principal do Capítulo~\ref{cap:estudo4}.
\label{alg:gzcmd-arranjo}

\noindent\rule{\textwidth}{0.4pt}
\begin{algorithmic}[1]
\REQUIRE Potenciais pares $P$, escore OpenRecLink, passo de blocagem, atributos do classificador, custos e orçamentos da v3.1
\ENSURE Aceitações acima do corte e decisões registradas abaixo dele
\STATE \textbf{Fase 0: Corte por passo}
\STATE Aceitar os potenciais pares com escore maior ou igual ao corte do respectivo passo
\STATE Definir $P^{-}$ como os potenciais pares abaixo do corte
\STATE \textbf{Fase 1: Recalibração sem os \textit{labels} externos}
\STATE Particionar por REFREC; ajustar o XGBoost na base inteira de treinamento
\STATE Ajustar a isotônica apenas nas probabilidades internas fora da amostra de $P^{-}$ do treinamento
\STATE Aplicar a calibração à partição externa de $P^{-}$
\STATE Estimar as proporções de erro e inconclusivos do revisor por validação cruzada nas partições externas; congelá-las antes das chamadas novas
\STATE \textbf{Fases 2 a 6: Política abaixo do corte}
\STATE Aplicar as fases 2 a 6 do Algoritmo~\ref{alg:gzcmd-v31} somente a $P^{-}$, usando a probabilidade recalibrada também na guarda superior
\STATE Reutilizar respostas apenas quando coincidirem modo, potencial par, modelo, instruções e dossiê
\STATE Manter os inconclusivos do LLM como encaminhamentos humanos pendentes
\RETURN Aceitações do corte e decisões de $P^{-}$, sem imputar acertos humanos
\end{algorithmic}

\refstepcounter{algorithm}%
\noindent\textbf{Algoritmo~\thealgorithm}: desenho avaliado do GZ-CMD com alternativa de calibração por conjuntos âncora, guardas e triagem da política v3, conforme \path{gzcmd/guardrails.py} e \path{gzcmd/gzcmd_v3_policy_engine.py}, e registro de decisões, sem exercício independente de auditoria das justificativas. A revisão histórica usou \textit{Platt scaling} supervisionado da nota; o Experimento~1 usou Random Forest calibrado por sigmoide. Essas execuções não usaram a Fase~1 por âncoras. A versão v3.1 é descrita no Algoritmo seguinte. As etapas de margem estreita e cardinalidade foram previstas, mas não implementadas e não integram as etapas abaixo (Seção~\ref{sec:e4-regras-guarda}).
\label{alg:gzcmd-pipeline}

\noindent\rule{\textwidth}{0.4pt}
\begin{algorithmic}[1]

\REQUIRE \parbox[t]{0.85\linewidth}{Conjunto de potenciais pares $P$ com escores brutos $\{s_i\}$;
  parâmetros $(C_{FP}, C_{FN}, C_{LLM}, e_{FP}, e_{FN})$;
  conjuntos âncora $A^+$ e $A^-$; orçamento de revisão $B$.}
\ENSURE \parbox[t]{0.85\linewidth}{Decisão $d_i \in \{\textsc{match}, \textsc{nonmatch}, \textsc{review}\}$ para cada par $i$,
  com campos de registro $(\mathit{razao}_i, \mathit{justificativa}_i)$.}

\medskip
\STATE \textbf{Fase 1: Calibração por âncoras}
\STATE Estimar $(\alpha,\beta)$ impondo $\mathbb{E}[\sigma(\alpha s + \beta) \mid s \in A^+] = r^+$
\STATE Estimar $(\alpha,\beta)$ impondo $\mathbb{E}[\sigma(\alpha s + \beta) \mid s \in A^-] = r^-$
\FORALL{par $i \in P$}
    \STATE $p_i \leftarrow \sigma(\alpha\, s_i + \beta)$
\ENDFOR

\medskip
\STATE \textbf{Fase 2: Regras de guarda determinísticas (versão v3)}
\FORALL{par $i \in P$}
    \STATE Inicializar indicador de revisão forçada $f_i \leftarrow 0$
    \IF{$(\mathit{data\_obito}_i - \mathit{data\_diagnostico}_i) < -180$ dias, com ambas as datas disponíveis}
        \STATE $d_i \leftarrow \textsc{always\_nonmatch}$; \quad $\mathit{razao}_i \leftarrow \texttt{temporal\_filter}$
        \STATE \textbf{continuar} para o próximo par
    \ENDIF
    \IF{nota final do CdR $v_i < 3$}
        \STATE $d_i \leftarrow \textsc{always\_nonmatch}$; \quad \textbf{continuar} para o próximo par
    \ENDIF
    \IF{$v_i \geq 7$ \AND indicadores de nascimento e endereço nulos \AND diferença conhecida de anos de nascimento $>5$}
        \STATE $f_i \leftarrow 1$; \quad $\mathit{razao}_i \leftarrow \texttt{homonimia\_risk}$ \COMMENT{indicadores e tratamento de faltantes da Seção~\ref{sec:e4-regras-guarda}}
        \STATE \textbf{continuar} para o próximo par
    \ENDIF
    \IF{$v_i \geq 10$ \AND concordância forte de nome, nascimento e município}
        \STATE $d_i \leftarrow \textsc{always\_match}$ \COMMENT{critérios de \path{gzcmd/guardrails.py}}
    \ENDIF
\ENDFOR

\medskip
\STATE \textbf{Fase 3: \textit{Expected loss engine}}
\FORALL{par $i$ sem guarda de aceitação ou rejeição}
    \STATE $L_M \leftarrow (1-p_i)\cdot C_{FP}$; \quad $L_N \leftarrow p_i\cdot C_{FN}$
    \STATE $L_R \leftarrow C_{LLM} + (1-p_i)\,e_{FP}\,C_{FP} + p_i\,e_{FN}\,C_{FN}$
    \STATE $\mathit{EVR}_i \leftarrow \min(L_M, L_N) - L_R$
    \IF{$L_M \leq L_N$}
        \STATE $d_i \leftarrow \textsc{match}$
    \ELSE
        \STATE $d_i \leftarrow \textsc{nonmatch}$
    \ENDIF
    \STATE Usar $(t_M,t_N)=(0{,}85;0{,}15)$ na vigilância e $(0{,}95;0{,}10)$ na confirmação
    \STATE $c_i \leftarrow (d_i=\textsc{match}\ \AND\ p_i<t_M)\ \OR\ (d_i=\textsc{nonmatch}\ \AND\ p_i>t_N)$
    \STATE Marcar pedido de revisão se $f_i=1$, $c_i=1$ ou $\mathit{EVR}_i>0$
    \STATE Registrar $(L_M, L_N, L_R, \mathit{EVR}_i)$ e código de razão preliminar
\ENDFOR

\medskip
\STATE \textbf{Fase 4: Triagem sob restrição de orçamento}
\STATE Excluir da seleção as guardas de aceitação e rejeição
\STATE Selecionar primeiro os pares com $f_i=1$, depois os com $c_i=1$, depois os com $\mathit{EVR}_i>0$, sem selecionar o mesmo par novamente
\STATE Dentro de cada prioridade, ordenar por $\mathit{EVR}_i$ decrescente até esgotar o orçamento restante $B$
\STATE Atribuir $d_i\leftarrow\textsc{review}$ apenas aos selecionados; os excedentes conservam a decisão automática de base, sem código adicional de razão

\medskip
\STATE \textbf{Fase 5: Revisão assistida por LLM em protocolo dual}
\FORALL{par $i$ com $d_i = \textsc{review}$}
    \STATE Montar dossiê JSON com subescores, $p_{\mathit{cal}}$ e códigos de guarda \COMMENT{dossiê com as saídas do motor; a v3.1 usa dossiê sem elas}
    \STATE Submeter ao Agente A e ao Agente B em chamadas separadas, com o mesmo modelo e dossiê
    \IF{há consenso entre os dois agentes}
        \STATE Adotar decisão consensual como decisão final
    \ELSE
        \STATE Encaminhar ao Agente Árbitro para deliberação final
    \ENDIF
    \STATE Registrar código de razão, justificativa textual, versão do modelo e carimbo temporal
\ENDFOR

\medskip
\STATE \textbf{Fase 6: Consolidação e retorno com registro de decisões}
\RETURN $\{(d_i, \mathit{razao}_i, \mathit{justificativa}_i)\}$ para todo par $i \in P$

\end{algorithmic}

% Algoritmo da política v3.1 (quatro ações)
\refstepcounter{algorithm}%
\noindent\textbf{Algoritmo~\thealgorithm}: política v3.1 do GZ-CMD, com probabilidades OOF agrupadas por REFREC, guardas da nota, quatro ações por \textit{expected loss}, revisão assistida por LLM com dossiê sem as saídas do motor e encaminhamento humano dos inconclusivos.
\label{alg:gzcmd-v31}

\noindent\rule{\textwidth}{0.4pt}
\begin{algorithmic}[1]

\REQUIRE \parbox[t]{0.85\linewidth}{Potenciais pares $P$ com nota final $v_i$ e 29 subescores $x_i$; \textit{labels} apenas no treinamento;
  custos $(C_{FP}, C_{FN}, C_H, C_{LLM})$; erro humano hipotético $(e_{FP,h}, e_{FN,h})$;
  proporções de erro e de inconclusivos do revisor LLM por modo e faixa $(e_{FP}, e_{FN}, u)$; orçamentos de revisão LLM e humana.}
\ENSURE \parbox[t]{0.85\linewidth}{Ação $d_i \in \{\textsc{match}, \textsc{nonmatch}, \textsc{llm}, \textsc{humano}\}$ para cada par $i$;
  decisão final do LLM $\in \{\textsc{match}, \textsc{nonmatch}, \textsc{unsure}\}$, com \textsc{unsure} encaminhado ao revisor humano.}

\medskip
\STATE \textbf{Fase 1: Probabilidade fora da amostra}
\STATE Particionar $P$ em cinco partições externas agrupadas por REFREC (semente 42); em cada uma, cinco partições internas, também agrupadas
\STATE Ajustar o XGBoost nas partições internas e a regressão isotônica sobre suas probabilidades fora da amostra
\STATE Reajustar o XGBoost em todo o treinamento externo e pontuar o teste externo
\STATE $p_i \leftarrow$ calibração isotônica do escore desse modelo no teste externo, sem acesso aos \textit{labels} desse teste

\medskip
\STATE \textbf{Fase 2: Guardas da nota}
\FORALL{par $i \in P$}
    \IF{$v_i < 5$}
        \STATE $d_i \leftarrow \textsc{always\_nonmatch}$
    \ELSIF{$v_i \geq 9$ \AND $p_i \geq 0{,}9$}
        \STATE $d_i \leftarrow \textsc{always\_match}$
    \ENDIF
\ENDFOR

\medskip
\STATE \textbf{Fase 3: Perdas das quatro ações}
\FORALL{par $i$ sem guarda}
    \STATE $L_M \leftarrow (1-p_i)\,C_{FP}$; \quad $L_N \leftarrow p_i\,C_{FN}$
    \STATE $L_H \leftarrow C_H + p_i\,e_{FN,h}\,C_{FN} + (1-p_i)\,e_{FP,h}\,C_{FP}$
    \STATE $L_L \leftarrow C_{LLM} + p_i\,e_{FN}\,C_{FN} + (1-p_i)\,e_{FP}\,C_{FP} + u\,L_H$
    \STATE $d_i \leftarrow \arg\min\{L_M, L_N, L_H, L_L\}$, com desempate na ordem vínculo, não-vínculo, humano e LLM
\ENDFOR

\medskip
\STATE \textbf{Fase 4: Orçamento}
\STATE Priorizar os pares com $d_i = \textsc{llm}$ pelo ganho sobre a melhor alternativa não LLM; depois, os demais pelo ganho da revisão humana
\IF{capacidade humana esgotada}
    \STATE Usar o LLM se houver capacidade e ganho sobre a decisão automática; caso contrário, aplicar a melhor ação automática
\ENDIF

\medskip
\STATE \textbf{Fase 5: Revisão assistida por LLM}
\FORALL{par $i$ com $d_i = \textsc{llm}$}
    \STATE Montar dossiê com 14 dos 29 subescores, nota e MACD, sem as saídas do motor ($p_{\mathit{cal}}$, faixa, EVR e códigos de guarda)
    \STATE Submeter aos dois agentes (instruções v4) e, se discordarem, ao árbitro
    \IF{decisão final $= \textsc{unsure}$}
        \STATE Encaminhar o par ao revisor humano
    \ENDIF
\ENDFOR

\medskip
\STATE \textbf{Fase 6: Retorno com registro de decisões}
\RETURN $\{(d_i, \text{decisão final}, \text{código de razão})\}$ para todo par $i \in P$

\end{algorithmic}
